krogan_clothes_01 = {
	default = "gfx/models/portraits/btr_09_krogan/krogan_empty_clothes.dds"
	
	game_setup = {#will run with a limited country scope. species and government is set but the country does not actually exist
		default = "gfx/models/portraits/btr_09_krogan/krogan_empty_clothes.dds"
	}
	
	#species scope
	species = { #generic portrait for a species
		default = "gfx/models/portraits/btr_09_krogan/krogan_empty_clothes.dds"
		
		random = {
			list = {
				"gfx/models/portraits/btr_09_krogan/krogan_empty_clothes.dds"
			}
		}
	}
	
	#pop scope
	pop_group = { #for a specific pop
		default = "gfx/models/portraits/btr_09_krogan/krogan_empty_clothes.dds"
		random = {
			list = {
				"gfx/models/portraits/btr_09_krogan/krogan_empty_clothes.dds"
			}
		}
	}

	#leader scope
	leader = { #scientists, generals, admirals, official
		default = "gfx/models/portraits/btr_09_krogan/krogan_empty_clothes.dds"

		random = {
			trigger = { 
				OR = {
					leader_class = commander
					leader_class = scientist
					leader_class = official
				}
			}
			list = {
				"gfx/models/portraits/btr_09_krogan/krogan_empty_clothes.dds"
			}
		}
	}

	#leader scope 
	ruler = { #for rulers
		default = "gfx/models/portraits/btr_09_krogan/krogan_empty_clothes.dds"
		
		# All Governments type
		random = {
			list = {
				"gfx/models/portraits/btr_09_krogan/krogan_empty_clothes.dds"
			}
		}
	}
}
