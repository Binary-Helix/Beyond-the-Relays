btr_menu_prev = {
	potential = { exists = from }
	allow = { always = yes }
	effect = {
		hidden_effect = {
			from = {
				if = {
					limit = { has_country_flag = btr_menu_page_00 }
					remove_country_flag = btr_menu_page_00
					set_country_flag = btr_menu_page_dev_news
					country_event = { id =  btr_datapad_menu.7 }
				}
				else_if = {
					limit = { has_country_flag = btr_menu_page_empire_dossier }
					remove_country_flag = btr_menu_page_empire_dossier
					set_country_flag = btr_menu_page_00
					country_event = { id =  btr_datapad_menu.1 }
				}
				else_if = {
					limit = { has_country_flag = btr_menu_page_economy }
					remove_country_flag = btr_menu_page_economy
					set_country_flag = btr_menu_page_empire_dossier
					country_event = { id =  btr_datapad_menu.2 }
				}
				else_if = {
					limit = { has_country_flag = btr_menu_page_relay_network }
					remove_country_flag = btr_menu_page_relay_network
					set_country_flag = btr_menu_page_economy
					country_event = { id =  btr_datapad_menu.3 }
				}
				else_if = {
					limit = { has_country_flag = btr_menu_page_warfare }
					remove_country_flag = btr_menu_page_warfare
					set_country_flag = btr_menu_page_relay_network
					country_event = { id =  btr_datapad_menu.4 }
				}
				else_if = {
					limit = { has_country_flag = btr_menu_page_planet_classes }
					remove_country_flag = btr_menu_page_planet_classes
					set_country_flag = btr_menu_page_warfare
