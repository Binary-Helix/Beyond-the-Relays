    }

        NOT = { has_planet_flag = planetary_station }
        owner = {
            has_tradition = tr_btr_anarchy_adoption
        }
    }

    allow = {
        hidden_trigger = {
            OR = {
                owner = { is_ai = no }
                AND = {
                    NOT = { has_forbidden_jobs = enforcer }
                    NOT = { has_available_jobs = enforcer }
                }
            }
        }
    }

    destroy_trigger = {
        exists = owner
        OR = {
            NOT = {	owner = { is_regular_empire = yes }	}
            AND = {
                owner = { is_ai = yes } #Players dont like random buildings blowing up
                planet = { has_building = building_hall_judgment }
            }#This way the AI gets to reclaim its building slots that it sunk into this expense
        }
    }

    planet_modifier = {
        job_enforcer_add = 200
    }

    resources = {
        category = planet_buildings
        cost = {
            minerals = 400
        }
        upkeep = {
            energy = 15
        }
        produces = {
            trigger = {
                owner = { has_technology = tech_asari_architecture }
            }
            unity = 0.2
        }
    }
    triggered_planet_modifier = {
        potential = {
            owner= {	has_technology = tech_asari_architecture }
        }
        modifier = {
            planet_amenities_add = 100
        }
    }

    prerequisites = {
        "tech_planetary_government"
    }

    triggered_desc = {
        text = job_enforcer_effect_desc
    }

    show_tech_unlock_if = {
        is_regular_empire = yes
        has_origin = omega_emp_origin
    }
}
