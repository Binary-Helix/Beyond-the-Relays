#Planet Meltification
tech_planet_meltification = {
	cost = @tier4cost2
	area = society
	tier = 4
	category = { new_worlds }
	prerequisites = { "tech_climate_restoration" }
	weight = @tier4weight2

	feature_flags = {
		terraform_frozen
	}

	weight_modifier = {
		modifier = {
			factor = 1.25
			has_tradition = tr_expansion_adopt
		}
		modifier = {
			factor = 0
			OR = {
				has_trait = trait_pc_ocean_preference
				has_trait = trait_robot_aquatic
				has_trait = trait_auto_hab_preference
			}
		}
		modifier = {
			factor = 0
			NOT = { has_global_flag = has_real_terraforming_mod }
		}
	}

	ai_weight = {
		factor = @terraforming_tech_ai_factor
		
		modifier = {
			factor = 2
			OR = {
				is_xenophobe = yes
				is_hive_empire = yes
			}
		}
		modifier = {
			factor = 5
			OR = {
				any_owned_planet = {
					habitability = {
						who = root
						value < 0.5
					}
				}
				any_system_within_border = {
					any_system_planet = {
						is_planet_class = pc_nuked
					}
				}
			}
		}
		modifier = {
			factor = 5
			has_origin = origin_life_seeded
		}
		modifier = {
			factor = 2
			is_ai = no #auto-research
		}
	}
}
