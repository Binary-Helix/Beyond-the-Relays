r_btr_ancient_generator = {
    activation_duration = @triumph_duration
    portrait = "#TODO"
    sound = "relic_activation_ether_drake_trophy"

    resources = {
        category = relics
        # Activation cost
        cost = {
            minerals = @activation_cost
        }
    }

    ai_weight = {
        weight = 100
    }

    triggered_country_modifier = {
        potential = {
            always = yes
        }
        country_energy_produces_mult = 0.1
    }

    score = 1000

    active_effect = {
        add_modifier = {
            modifier = "ancient_generator"
            months = 5
        }
        #custom_tooltip = relic_triumph_cooldown
        hidden_effect = {
            add_modifier = {
                modifier = "relic_activation_cooldown"
                months = 5
            }
        }
    }

    # Possible check for activation
    possible = {
        inline_script = {
            script = relics/activation_checks
            RELIC = r_ancient_generator
        }
    }
}
