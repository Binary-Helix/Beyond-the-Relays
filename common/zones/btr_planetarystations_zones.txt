		}
	}

	unlock = { # planet scope

			is_wilderness_empire = no
		}
	}

		category = planet_zones
		cost = {

		}
	}

		zone_industrial
	}
	
	zone_sets = {
		station_foundry
	}

		foundry
		automation
		origin
	}

	inline_script = {
		script = jobs/zone_foundry_add
		AMOUNT = @scaling_district_1_job
	}

	planet_modifier = {
		zone_building_slots_add = 2
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_tech_or_functional_civic = {
					TECH = tech_colonial_station_protocol
				}
			}
		}
		zone_building_slots_add = 1
    }
}
