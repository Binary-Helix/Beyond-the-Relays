decision_resort_station = {
	owned_planets_only = yes
	sound = event_construction
	icon = decision_resort_station
	
	enactment_time = 150
	resources = {
		category = decisions
		cost = {
			energy = 500
			unity = 500
		}
	}
	
	potential = {
		has_planet_flag = planetary_station
		exists = owner
		owner = { 
			is_gestalt = no 
		}
		NOT = { has_modifier = resort_station }
	}
	
	allow = {
		custom_tooltip = {
			fail_text = other_station_type
			is_specific_station_type = no
		}
		custom_tooltip = {
			fail_text = bad_place_to_rest
			NOR = {
				has_mining_deposit = yes
				has_energy_deposit = yes
			}
		}
	}
	
	abort_trigger = {
		OR = {
			owner = { is_gestalt = yes }
			has_modifier = prison_station
			has_modifier = military_outpost
			has_modifier = resort_station
		}
 	}
	
	effect = {
		hidden_effect = {
			custom_tooltip = district_casino_level_add
			custom_tooltip = district_race_track_add
			
			add_modifier = {
				modifier = resort_station
				days = -1
			}
			
			validate_planet_buildings_and_districts = yes
		}
	}
	
	ai_weight = {
		weight = 5
		modifier = {
			factor = 0.5
			owner = {
				any_owned_planet = {
					has_modifier = resort_station
				}
			}
		}
		modifier = {
			factor = 0
			solar_system = {
				any_system_planet = {
					has_modifier = resort_station
				}
			}
		}
	}
}

	owned_planets_only = yes
	sound = event_construction
	icon = decision_crime
	
	enactment_time = 90
	resources = {
		category = decisions
		cost = {
			unity = 1000
		}
	}
	
	potential = {
		owner = { 
			any_owned_planet = { has_modifier = prison_station }
		}
		OR = {
			AND = {
				has_modifier = anticrime_campaign
				any_owned_pop_job = { has_job_type = criminal }
			}
			AND = {
				has_modifier = political_persecution
				any_owned_pop_group = { has_ethic = ethic_egalitarian }
			}
			AND = {
				has_modifier = hunting_heretics
				any_owned_pop_group = { has_ethic = ethic_materialist }
			}
		}
		any_owned_pop_job = { has_job_type = enforcer }
	}
	
	allow = {
		is_controlled_by = owner
		has_ground_combat = no
		has_orbital_bombardment = no
	}

		hidden_effect = {
			owner = {
				random_owned_planet = { 
					limit = { 
						has_modifier = prison_station
						OR = {
							pop_amount < 3000
							NOT = { has_modifier = planet_population_control }
						}
					}
					save_event_target_as = prison_station
				}
			}
			
			if = {
				limit = { has_modifier = political_persecution }
				custom_tooltip = decision_detention_egalitarian_effect_desc
				weighted_random_owned_pop_group = {
					limit = { has_ethic = ethic_egalitarian }
					set_timed_pop_group_flag = {
						flag = pop_convicted
						years = 20
					}
					resettle_pop_group = {
						pop = this
						planet = event_target:prison_station
					}
				}
			}
			else_if ={
				limit = { has_modifier = hunting_heretics }
				custom_tooltip = decision_detention_materialist_effect_desc
				weighted_random_owned_pop_group = {
					limit = { has_ethic = ethic_materialist }
					set_timed_pop_group_flag = {
						flag = pop_convicted
						years = 20
					}
					resettle_pop_group = {
						pop = this
						planet = event_target:prison_station
					}
				}
			}
			else = {
				custom_tooltip = decision_detention_criminal_effect_desc
				weighted_random_owned_pop_group = {
					limit = {
						is_pop_category = criminal
					}
					set_timed_pop_group_flag = {
						flag = pop_convicted
						years = 20
					}
					resettle_pop_group = {
						pop = this
						planet = event_target:prison_station
					}
				}
			}
			
			owner = {
				create_message = {
					type = FELONY_TRANSPORT
					localization = FELONY_TRANSPORT_DESC
					days = 30
					target = root
					variable = {
						type = name
						localization = PLANET
						scope = root
					}
					variable = {
						type = name
						localization = PRISON
						scope = event_target:prison_station
					}
				}
			}
		}
	}
	
	ai_weight = {
		weight = 20
	}
}
