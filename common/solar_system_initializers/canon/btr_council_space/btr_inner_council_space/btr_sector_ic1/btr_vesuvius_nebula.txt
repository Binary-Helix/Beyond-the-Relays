sys_pompeii_init = {
    name = "Pompeii"
    class = sc_g
	max_instances = 1
    usage = custom_empire

    flags = { inner_council sector_ic1 vesuvius_nebula pompeii_system secondary_relay_system }

    init_effect = {
        generate_btr_home_system_resources = yes
    }

    planet = {
	    name = "Pompeii"
        class = pc_g_star
        orbit_distance = 0
        size = 20
        flags = { pompeii_star }
    }
	
	planet = {
		name = "Rakhana"
		class = pc_desert
		orbit_distance = 150
		size = 18
        starting_planet = yes
		entity = "mephitic_desert_planet_12_entity"
		flags = { rakhana_planet subclass_world }
        deposit_blockers = none
        modifiers = none
		init_effect = {
            add_modifier = {
                modifier = dust_world
                days = -1
            }
            save_global_event_target_as = drellHomeworld
			if = {
				limit = {
					exists = event_target:drellHomeworld
				}
				log = "Drell Homeworld Event Target Set"
			}
            prevent_anomaly = yes

            check_for_drell_player = yes

            if = {
				limit = {
					any_country = {
						has_country_flag = remnants_of_rakhana_player
					}
				}
				random_country = {
					limit = {
						has_country_flag = remnants_of_rakhana_player
					}
					save_global_event_target_as = remnants_of_rakhana
				}
                generate_empire_home_planet = yes
			}
			else = {
				generate_remnants_of_rakhana_country = yes
			}
            set_capital = yes
			set_owner = event_target:remnants_of_rakhana
			create_colony = {
				owner = event_target:remnants_of_rakhana
				species = event_target:drellSpecies
				ethos = owner
			}
            generate_btr_start_deposits_and_blockers = yes
            btr_generate_canon_planetary_features = yes
            inline_script = "game_start/canon/remnants_rak_origin"
            generate_dre_rakhana_pops = yes
            event_target:remnants_of_rakhana = {
                create_starting_leaders = yes
                country_event = { id = game_start.9 }
            }
		}
	}

	planet = {
		name = "Kahtack"
		class = pc_iron
		orbit_distance = 150
		size = 8
		flags = { kahtack_planet }
	}

	planet = {
		name = "Erusa"
		class = pc_gas_giant
		orbit_distance = 150
		size = 34
		flags = { erusa_planet }
	}	

	init_effect = { 
		spawn_megastructure = { 
			type = "inner_council_sector_1_secondaryrelay" 
			name = "Sector IC1 - Secondary Relay 2"
			orbit_angle = 270 
			orbit_distance = 500
		} 
	}
}
