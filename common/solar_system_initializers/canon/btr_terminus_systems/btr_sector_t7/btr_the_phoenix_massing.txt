sys_the_sea_of_storms_init = {
    name = "The Sea of Storms"
    class = sc_deep_space
	max_instances = 1
    usage = custom_empire

    flags = { terminus_systems sector_t7 the_phoenix_massing the_sea_of_storms_system }

    init_effect = {
        save_global_event_target_as = nazaraFirstFleet
    }

	planet = {
		class = pc_invisible_star
		size = 30
	}

	change_orbit = 90

    planet = {
        name = "Heretic Station"
        class = pc_heretic_station
        orbit_distance = 0
        orbit_angle = 144
        size = 22
        has_ring = no   
        starting_planet = yes
        flags = {heretic_station heretic_station_planet}
        deposit_blockers = none
        modifiers = none
        init_effect = {
			save_global_event_target_as = disciplesHomeStation

            prevent_anomaly = yes

            check_for_geth_player = yes

            if = {
				limit = {
					any_country = {
						has_country_flag = disciples_of_nazara_player
					}
				}
				random_country = {
					limit = {
						has_country_flag = disciples_of_nazara_player
					}
					save_global_event_target_as = disciples_of_nazara
				}
                generate_empire_home_planet = yes
            }
			else = {
				generate_disciples_of_nazara_country = yes
			}
            set_capital = yes
			set_owner = event_target:disciples_of_nazara
			create_colony = {
				owner = event_target:disciples_of_nazara
				species = event_target:gethSpecies
				ethos = owner
			}
            generate_btr_start_deposits_and_blockers = yes
            btr_generate_canon_planetary_features = yes
            inline_script = "game_start/canon/disciples_naz_origin"
            generate_naz_station_pops = yes
            event_target:disciples_of_nazara = {
                create_starting_leaders = yes
                country_event = { id = game_start.9 }
            }
        }
    }
}
