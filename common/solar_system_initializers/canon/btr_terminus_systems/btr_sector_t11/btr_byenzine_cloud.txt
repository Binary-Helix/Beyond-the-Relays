sys_taesin_init = {
    name = "Taesin"
    class = sc_g
	max_instances = 1
    usage = custom_empire

	flags = { terminus_systems sector_t11 byenzine_cloud taesin_system }

    init_effect = {
        save_global_event_target_as = lystheniFirstFleet
    }

    planet = {
	    name = "Taesin"
        class = pc_g_star
        orbit_distance = 0
        size = 20
        flags = { taesin_star }
    }

    planet = {
        name = "Lorel"
        class = pc_hothouse
        orbit_distance = 150
        orbit_angle = 92
        has_ring = no
    }

    planet = {
        name = "Sepkai"
        class = pc_tropical
        orbit_distance = 150
        orbit_angle = 62
        has_ring = no
        starting_planet = yes
		flags = { sepkai_planet subclass_world }
		entity = "biolum_planet_08_entity"	
        deposit_blockers = none
        modifiers = none
		init_effect = {
            add_modifier = {
                modifier = biolum_world
                days = -1
            }
            save_global_event_target_as = lystheniHomeworld
			if = {
				limit = {
					exists = event_target:lystheniHomeworld
				}
				log = "Lystheni Homeworld Event Target Set"
			}
            prevent_anomaly = yes

            check_for_lystheni_player = yes

            if = {
				limit = {
					any_country = {
						has_country_flag = lystheni_order_player
					}
				}
				random_country = {
					limit = {
						has_country_flag = lystheni_order_player
					}
					save_global_event_target_as = lystheni_order
				}
                generate_empire_home_planet = yes
			}
			else = {
				generate_lystheni_order_country = yes
			}
            set_capital = yes
			set_owner = event_target:lystheni_order
			create_colony = {
				owner = event_target:lystheni_order
				species = event_target:lystheniSpecies
				ethos = owner
			}
            generate_btr_start_deposits_and_blockers = yes
            btr_generate_canon_planetary_features = yes
            inline_script = "game_start/canon/lystheni_ord_origin"
            generate_lys_sepkai_pops = yes
            event_target:lystheni_order = {
                create_starting_leaders = yes
                country_event = { id = game_start.9 }
            }
		}

		moon = {
			name = "Nonau"
			class = pc_barren
			orbit_distance = 20
			size = 7	
			flags = { nonau_planet }
		}

		moon = {

    }
