sys_nosikae_init = {
    name = "Nosikae"
    class = sc_k_giant
	max_instances = 1
    usage = custom_empire

    flags = { terminus_systems sector_t6 eyes_of_mizhuna nosikae_system secondary_relay_system }	

    init_effect = {
        save_global_event_target_as = collectivefirstFleet
    }

    planet = {
		name = "Nosikae"
        class = pc_k_giant_star
        orbit_distance = 0
        size = 35
        flags = { nosikae_star }
    }

	planet = {
		name = "Sed'riicae"
		class = pc_dextro_desert
		orbit_distance = 125
		size = 24
		has_ring = no
        starting_planet = yes
        flags = { sedriicae_planet subclass_world }
        deposit_blockers = none
        modifiers = none
		init_effect = {
            add_modifier = {
                modifier = dextro_world
                days = -1
            }
            save_global_event_target_as = forayanHomeworld
			if = {
				limit = {
					exists = event_target:forayanHomeworld
				}
				log = "Forayan Homeworld Event Target Set"
			}
            prevent_anomaly = yes

            check_for_forayan_player = yes

            if = {
				limit = {
					any_country = {
						has_country_flag = sedriicae_collective_player
					}
				}
				random_country = {
					limit = {
						has_country_flag = sedriicae_collective_player
					}
					save_global_event_target_as = sedriicae_collective
				}
                generate_empire_home_planet = yes
			}
			else = {
				generate_sedriicae_collective_country = yes
			}
            set_capital = yes
			set_owner = event_target:sedriicae_collective
			create_colony = {
				owner = event_target:sedriicae_collective
				species = event_target:forayanSpecies
				ethos = owner
			}
            clear_blockers = yes
            generate_btr_start_deposits_and_blockers = yes
            btr_generate_canon_planetary_features = yes
            inline_script = "game_start/canon/sedriicae_col_origin"
            generate_sed_sedriicae_pops = yes
            event_target:sedriicae_collective = {
                create_starting_leaders = yes
                country_event = { id = game_start.9 }
            }
		}
	}
	
	planet = {
		name = "Xhentura"
		class = pc_barren
		orbit_distance = 125
		size = 10
		flags = { xhentura_planet }
		has_ring = no
	}
	
	planet = {
		name = "Iciinae"
		class = pc_hothouse
		orbit_distance = 125
		size = 20
		flags = { iciinae_planet }
		has_ring = yes	
	}

	planet = {
		name = "Lesiik"
		class = pc_frozen
		orbit_distance = 125
		size = 12
		flags = { lesiik_planet }
		has_ring = no	
	}

	init_effect = { 
		spawn_megastructure = { 
			type = "terminus_sector_6_secondaryrelay" 
			name = "Sector T6 - Secondary Relay 1"
			orbit_angle = 270

		}

}
