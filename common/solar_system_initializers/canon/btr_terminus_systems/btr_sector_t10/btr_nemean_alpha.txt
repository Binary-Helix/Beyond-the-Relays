sys_cheng_init = {
	name = "Cheng"
	class = sc_m_hyper
	max_instances = 1

    flags = { terminus_systems sector_t10 nemean_alpha cheng_system primary_relay_system planet_state }

	planet = {
		name = "Cheng"
		class = pc_m_hyper_star
		size = 35
		flags = { cheng_star }
	}

	planet = {
		name = "Jenera"
		class = pc_dextro_savannah
        orbit_distance = 250
		size = 17
		entity = "aquifer_planet_03_entity"
        deposit_blockers = none
        modifiers = none
		flags = { subclass_world jenera_planet }
		init_effect = {
            save_global_event_target_as = jeneraHomeworld
            if = {
                limit = {
                    exists = event_target:jeneraHomeworld
                }
                log = "Jenera Corporation Event Target Set"
            }
            prevent_anomaly = yes

            generate_jenera_corporation_country = yes

            set_capital = yes
			set_owner = event_target:jenera_corporation
			create_colony = {
				owner = event_target:jenera_corporation
				species = event_target:batarianSpecies
				ethos = owner
			}

            generate_jenera_pops = yes

            add_modifier = {
				modifier = aquifer_world
				days = -1
			}

            add_deposit = d_eezo_trace
        }
	}

	init_effect = { 
		spawn_megastructure = { 
			type = "nemeanalpha_mouthoffandrar_primaryrelay" 
			name = "Sector T10 - Primary Relay 1"
			orbit_angle = 270 
			orbit_distance = 500
		} 
	}
}
