generate_asari_republic_playtest_setup = {
    add_resource = { energy = 10000	}
    add_resource = { minerals = 10000 }
    add_resource = { food = 10000 }
    add_resource = { consumer_goods = 10000 }
    add_resource = { light_alloys = 10000	}
    add_resource = { beryllium = 10000 }
    add_resource = { lithium = 10000 }
    add_resource = { titanium = 10000 }
    add_resource = { heavy_alloys = 10000	}
    add_resource = { iridium = 10000 }
    add_resource = { palladium = 10000 }
    add_resource = { platinum = 10000 }
    add_resource = { antimatter_fuel = 10000 }
    add_resource = { helium_3 = 10000 }
    add_resource = { hydrogen = 10000 }
    add_resource = { uranium = 10000 }
    add_resource = { eezo = 10000 }
    add_resource = { trade = 10000 }
    add_resource = { influence = 100 }
    add_resource = { unity = 450 }

    log = "Asari Republics Starting Resources Set"

    add_modifier = {
        modifier = asari_rep_empire_modifier
        days = -1
    }

    add_modifier = {
        modifier = btr_citadel_council_member
        days = -1
    }

    set_country_flag = citadel_council_member

    log = "Asari Republics Empire Modifier Set"

    every_planet_within_border = {
        limit = {
            has_deposit_for = shipclass_mining_station
            has_mining_station = no
        }
        create_mining_station = {
            owner = ROOT
        }
    }
    every_planet_within_border = {
        limit = {
            has_deposit_for = shipclass_research_station
            has_research_station = no
        }
        create_research_station = {
            owner = ROOT
        }
    }

    log = "Asari Republics Mining and Research Stations Set"
}
