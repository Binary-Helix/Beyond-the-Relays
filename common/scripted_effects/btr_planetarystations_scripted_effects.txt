set_station_deposits = {
		
	# Mineral Deposits
	
	if = {
		limit = {
			OR = {
				has_modifier = junkyard
				has_modifier = ore_freighter
			}
		}
		add_deposit = d_junkyard
	}
	else_if = {
		limit = { has_planet_flag = minerals_1 }
		if = {
			limit = {
				OR = {
					is_gas_giant_station = yes
					is_ice_giant_station = yes
				}
			}
			add_deposit = d_ore_rich_rings
		}
		else = {
			add_deposit = d_mineral_lodes
		}
	}
	else_if = {
		limit = { has_planet_flag = minerals_2 }
		if = {
			limit = {
				OR = {
					is_gas_giant_station = yes
					is_ice_giant_station = yes
				}
			}
			add_deposit = d_ore_rich_rings
			add_deposit = d_ore_rich_rings
		}
		else = {
			add_deposit = d_mineral_lodes
			add_deposit = d_mineral_lodes
		}
	}
	else_if = {
		limit = { has_planet_flag = minerals_3 }
		if = {
			limit = {
				OR = {
					is_gas_giant_station = yes
					is_ice_giant_station = yes
				}
			}
			add_deposit = d_ore_rich_rings
			add_deposit = d_ore_rich_rings
			add_deposit = d_ore_rich_rings
		}
		else = {
			add_deposit = d_mineral_lodes
			add_deposit = d_mineral_lodes
			add_deposit = d_mineral_lodes
		}
	}
	else_if = {
		limit = { has_planet_flag = minerals_4 }
		if = {
			limit = {
				OR = {
					is_gas_giant_station = yes
					is_ice_giant_station = yes
				}
			}
			add_deposit = d_ore_rich_rings
			add_deposit = d_ore_rich_rings
			add_deposit = d_ore_rich_rings
			add_deposit = d_ore_rich_rings
		}
		else = {
			add_deposit = d_mineral_lodes
			add_deposit = d_mineral_lodes
			add_deposit = d_mineral_lodes
			add_deposit = d_mineral_lodes
		}
	}
	else = {
		limit = {
			OR = {
				has_planet_flag = minerals_5
				has_planet_flag = minerals_larg
			}
		}
		if = {
			limit = {
				OR = {
					is_gas_giant_station = yes
					is_ice_giant_station = yes
				}
			}
			add_deposit = d_ore_rich_rings
			add_deposit = d_ore_rich_rings
			add_deposit = d_ore_rich_rings
			add_deposit = d_ore_rich_rings
			add_deposit = d_ore_rich_rings
		}
		else = {
			add_deposit = d_mineral_lodes
			add_deposit = d_mineral_lodes
			add_deposit = d_mineral_lodes
			add_deposit = d_mineral_lodes
			add_deposit = d_mineral_lodes
		}
	}
	
	# Energy Deposits
	
	if = {
		limit = { has_modifier = precious_moon }
		add_deposit = d_precious_moon
	}
	else_if = {
		limit = { has_planet_flag = energy_1 }
		if = {
			limit = {
				OR = {
					is_planet_class = pc_molten_station
					is_planet_class = pc_sulfur_station
				}
			}
			add_deposit = d_geothermal_springs
		}
		else_if = {
			limit = { 
				OR = {
					is_barren_cold_station = yes
					is_asteroid_station = yes
				}
			}
			add_deposit = d_rich_regolith_formations
		}
		else_if = {
			limit = {
				OR = {
					is_gas_giant_station = yes
					is_ice_giant_station = yes
					is_planet_class = pc_hot_giant_station
				}
			}
			add_deposit = d_helium_clouds
		}
		else_if = {
			limit = { is_planet_class = pc_greenhouse_station }
			add_deposit = d_storm_area
		}
		else_if = {
			limit = { is_planet_class = pc_scorched_station }
			add_deposit = d_scorched_plain
		}
		else = {
			add_deposit = d_hydrocarbon_deposits
		}
	}
	else_if = {
		limit = { has_planet_flag = energy_2 }
		if = {
			limit = {
				OR = {
					is_planet_class = pc_molten_station
					is_planet_class = pc_sulfur_station
				}
			}
			add_deposit = d_geothermal_springs
			add_deposit = d_geothermal_springs
		}
		else_if = {
			limit = { 
				OR = {
					is_barren_cold_station = yes
					is_asteroid_station = yes
				}
			}
			add_deposit = d_rich_regolith_formations
			add_deposit = d_rich_regolith_formations
		}
		else_if = {
			limit = {
				OR = {
					is_gas_giant_station = yes
					is_ice_giant_station = yes
					is_planet_class = pc_hot_giant_station
				}
			}
			add_deposit = d_helium_clouds
			add_deposit = d_helium_clouds
		}
		else_if = {
			limit = { is_planet_class = pc_greenhouse_station }
			add_deposit = d_storm_area
			add_deposit = d_storm_area
		}
		else_if = {

		export_trigger_value_to_variable = {
			trigger = num_districts
			parameters = {
				type = district_habitable_dome
			}
			variable = num_housing_districts
		}
	}
	if = {
		limit = { has_district = district_habitable_complex }
		export_trigger_value_to_variable = {
			trigger = num_districts
			parameters = {
				type = district_habitable_complex
			}
			variable = num_housing_districts
		}
	}
	if = {
		limit = { has_district = district_cell_block }
		export_trigger_value_to_variable = {
			trigger = num_districts

				type = district_cell_block
			}
			variable = num_housing_districts
