

change_migrant_fleet_system = {
	every_owned_species = {
		limit = {
			is_species_class = QUA
		}
		save_global_event_target_as = quarianSpecies
	}
	every_system = {
		if = {
			limit = {
				has_star_flag = migrant_fleet_main_system
			}
			create_starbase = {
				size = starbase_outpost_quarian
				owner = event_target:migrant_fleet_player
			}
			every_system_planet = {
				if = {
					limit = {
						has_planet_flag = migrant_fleet_planet
					}
					set_owner = event_target:migrant_fleet_player
					create_colony = {
						owner = event_target:migrant_fleet_player
						species = event_target:quarianSpecies
					}
					event_target:migrant_fleet_capital = {
						set_capital = yes
						generate_empire_home_planet = yes
					}
				}
			}
			set_star_class = sc_a
		}
		if = {
			limit = {
				has_star_flag = migrant_fleet_spawn_system
			}
			save_global_event_target_as = quarianFleetsStart
			every_fleet_in_system = {
				destroy_fleet = this
			}
			set_name = "Botesh"
			every_system_planet = {
				if = {
					limit = {
						has_planet_flag = migrant_fleet_start
					}
					destroy_colony = yes
					change_pc = pc_barren_cold
					reroll_planet = yes
				}
			}
		}
	}
}
