# several equivalent jobs, such as Technicians and Technician Drones.
# Needs the scope valid checks and default fallback for e.g. galaxy setup, observer mode.

# Helium Extractors Icon
defined_text = {
	name = GetHeliumIcon
	text = {
		trigger = {
			is_scope_valid = yes
			planet = { has_deposit = d_rich_regolith_formations }
		}
		localization_key = job_regolith_miner_icon
	}
	text = {
		trigger = {
			is_scope_valid = yes
			planet = { has_deposit = d_helium_clouds }
		}
		localization_key = job_helium_extractor_icon
	}
	default = job_helium_extractor_icon
}
