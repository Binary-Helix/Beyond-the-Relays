namespace = btr_game_start

#####################
# MASTER START EVENTS
#####################
event = {
	id = btr_game_start.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_global_flag = game_started
		set_global_flag = has_beyond_the_relays
		log = "Playing Beyond the Relays"

		every_country = {
			limit = { is_btr_empire = yes }
			country_event = { id = btr_game_start.4 days = 0 } #Set starting Territory
			country_event = { id = btr_game_start.10 days = 2 } #Set Starting Contacts and Flags
			country_event = { id = btr_game_start.11 days = 1 } #Set Species Rights
		}

		### SPAWN INVISIBLE GLOBAL EVENT COUNTRY
		create_country = {
			name = "global_event_country"
			type = global_event
			effect = {
				save_global_event_target_as = global_event_country
			}
		}

	}
}

	hide_window = yes
	is_triggered_only = yes
	immediate = {
		log = "btr_game_start.11 - Started"
		every_country = {
			random_owned_pop_group = {
				limit = {
					has_pop_group_flag = init_spawn
					is_same_species = owner_species
				}
				save_event_target_as = realOwnerSpecies
			}
			if = {
				limit = { exists = event_target:realOwnerSpecies }
				change_dominant_species = { species = event_target:realOwnerSpecies	}
			}
		}
		every_country = {
			limit = {
				NOR = {
					has_country_flag = hegemony_of_kharshan
					has_country_flag = lystheni_order
					has_country_flag = omega_empire
					has_country_flag = collector_empire
					has_country_flag = cerberus_empire
					has_country_flag = disciples_of_nazara
					is_independent_colony = yes
				}
			}
			every_owned_pop_species = {
				limit = {
					NOT = {
						is_same_species = owner_species
					}
				}
				set_citizenship_type = { country = root type = citizenship_limited cooldown = yes }
				set_living_standard = { country = root type = living_standard_normal cooldown = yes }
			}
		}
		every_country = {
			limit = {
				OR = {
					has_country_flag = hegemony_of_kharshan

			}

				set_living_standard = { country = root type = living_standard_omega_subsistence cooldown = yes }
			}

				policy = slavery

			}
