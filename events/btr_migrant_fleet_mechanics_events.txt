	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_fleet_flag = quarianCivilianFleet
	}

	immediate = {
		root.owner = {
			set_migrant_fle_total_production_variables = yes
		}

			save_event_target_as = migrantfleetLocation
			every_system_planet = {
				limit = {
					is_primary_star = yes
				}

				change_migrant_fleet_system_star = yes
			}
			every_system_planet = {
				limit = {
					has_deposit = yes
				}
				set_migrant_fleet_planet_deposit_variables = yes
