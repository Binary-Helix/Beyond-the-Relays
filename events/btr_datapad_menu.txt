# On press begin - Beyond the Relays Datapad Initialization
country_event = {
	id = btr_datapad_menu.0

	fire_only_once = yes

		country_event = { id = btr_datapad_menu.001 }
	}

# On press begin - DERPLLECTOR CHECKS FOR INTEGRATED OR INCOMPATIBLE MODS AND CUSTOM EMPIRE
country_event = {
	id = btr_datapad_menu.001
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		if = {
			limit = {
				OR = {
					has_global_flag = beyond_the_advisors_installed
					has_global_flag = has_beyond_the_worlds_mod
					has_global_flag = has_real_space_mod
				}
			}
			set_global_flag = integrated_mods_used
			log = "Using Integrated Mods"
		}
		if = {
			limit = {
				has_global_flag = planetarydiversity
			}
			set_global_flag = incompatible_mods_used
			log = "Using Incompatible Mods"
		}
		if = {
			limit = {
				is_country_type = default
				is_ai = no
				NOT = {
					has_country_flag = 	btr_empire
				}
			}
			set_global_flag = custom_empire_used
			log = "Using Custom Empire"
		}

		country_event = { id = btr_datapad_menu.002 }
	}
}

# DERPLLECTOR BEGINS SNIFFING
country_event = {
	id = btr_datapad_menu.002
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		log = "Derpllector Hunts"
		#Check for only custom_empire_used.
		if = {
			limit = {
				NOT = {
					has_global_flag = incompatible_mods_used
					has_global_flag = integrated_mods_used
				}
				has_global_flag = custom_empire_used
			}
			country_event = { id = btr_datapad_menu.003 }
		}
		#Check for custom_empire_used and incompatible_mods.
		if = {
			limit = {
				NOT = {
					has_global_flag = integrated_mods_used
				}
				has_global_flag = custom_empire_used
				has_global_flag = incompatible_mods_used
			}
			country_event = { id = btr_datapad_menu.004 }
		}
		#Check for custom_empire_used and integrated_mods_used.
		if = {
			limit = {
				NOT = {
					has_global_flag = incompatible_mods_used
				}
				has_global_flag = custom_empire_used
				has_global_flag = integrated_mods_used
			}
			country_event = { id = btr_datapad_menu.005 }
		}
		#Check for only integrated_mods_used.
		if = {
			limit = {
				NOT = {
					has_global_flag = custom_empire_used
					has_global_flag = incompatible_mods_used
				}
				AND = {
					has_global_flag = integrated_mods_used
				}
			}
			country_event = { id = btr_datapad_menu.006 }
		}
		#Check for integrated_mods_used and incompatible_mods_used.
		if = {
			limit = {
				NOT = {
					has_global_flag = custom_empire_used
				}
				AND = {
					has_global_flag = integrated_mods_used
					has_global_flag = incompatible_mods_used
				}
			}
			country_event = { id = btr_datapad_menu.007 }
		}
		#Check for only incompatible_mods_used.
		if = {
			limit = {
				NOT = {
					has_global_flag = custom_empire_used
					has_global_flag = integrated_mods_used
				}
				has_global_flag = incompatible_mods_used
			}
			country_event = { id = btr_datapad_menu.008 }
		}
		#If custom_empire_used flag missing (indicating Custom Empire) and has BOTH integrated and incompatible mods flag, trigger Derpllector Sees All view.
		if = {
			limit = {
				AND = {
					has_global_flag = integrated_mods_used
					has_global_flag = custom_empire_used
					has_global_flag = incompatible_mods_used
				}
			}
			country_event = { id = btr_datapad_menu.009 }
		}
	}
}
