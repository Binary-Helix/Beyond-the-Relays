# Granted Council Spectre
country_event = {
    id = btr_council_resolutions.3
    title = "btr_council_resolutions.3.name"
    desc = "btr_council_resolutions.3.desc"
    picture = GFX_evt_granted_council_spectre
    show_sound = btr_event_granted_council_spectre

    is_triggered_only = yes

    option = {
        name = btr_council_resolutions.3.a
        add_modifier = {
            modifier = btr_granted_council_spectre
            days = -1
        }

        hidden_effect = {
            save_event_target_as = spectre_country
            set_country_flag = spectre_1_approved
            event_target:spectre_country  = {
                create_leader = {
                    class = btr_council_spectre
                    tier = leader_tier_renowned
                    species = owner_main_species
                    name = random
                    skill = 5
                    event_leader = no
                    custom_description = spectre_leader_desc
                    custom_catch_phrase = spectre_leader_catch_phrase
                    immortal = no
                    hide_age = no

                    randomize_traits = no
                    effect = {
                        set_skill = 5
                        set_leader_flag = is_in_recruit_window
                        save_event_target_as = new_spectre
                        add_trait = {
                            trait = btr_leader_trait_council_spectre
                        }
                    }
                }
                create_fleet = {
                    effect = {
                        set_owner = prev
                        create_ship = {
                            name = random
                            design = "NAME_Spectre_Vessel"
                            prefix = no
                            upgradable = yes
                        }
                        set_location = {
                            target = prev.capital_scope.star
                            distance = 200
                            angle = random
                        }
                        set_fleet_stance = passive
                        assign_leader = event_target:new_spectre
                        set_fleet_flag = council_spectre
                    }
                    settings = {
                        can_upgrade = yes
                        can_change_leader = no
                        uses_naval_capacity = no
                        spawn_debris = yes
                        can_disband = no
                    }
                }
            }
        }
    }
}
