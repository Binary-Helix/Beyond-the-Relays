namespace = planetarystations_terraforming_event


#Terraform Planetary Station
planet_event = {
	id = planetarystations_terraforming_event.1
	hide_window = yes
	
	is_triggered_only = yes
	
	immediate = {
		terraform_planetary_station = yes
	}
}

#Complete Barren Terraforming Project 1
planet_event = {
	id = planetarystations_terraforming_event.2
	hide_window = yes
	
	is_triggered_only = yes
	
	immediate = {
		if = {
			limit = { has_planet_flag = mars }
			set_planet_entity = {
				picture = pc_barren_terraformed_station
				entity = barren_terraformed_planet_mars_entity
				atmosphere_color = hsv { 0.50 0.3 0.7 }
				atmosphere_intensity = 1.0
				atmosphere_width = 0.5
			}
		}
		else = {
			set_planet_entity = {
				picture = pc_barren_terraformed_station
				entity = barren_terraformed_planet_01_entity
				atmosphere_color = hsv { 0.50 0.3 0.7 }
				atmosphere_intensity = 1.0
				atmosphere_width = 0.5
			}
		}
		add_modifier = { modifier = barren_terraformed days = -1 }
	}
}
