#Reference btr_game_start.11
country_event = {
	id = btr_independent_worlds.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
        #Every Independent World Main Species Citizenship and Standards
        random_country = {
            limit = {
                has_country_flag = btr_independent_colony
            }
            every_owned_pop_species = {
                limit = {
                    is_same_species = owner_species
                }
                set_citizenship_type = {
                    country = root
                    type = citizenship_full
                    cooldown = yes
                }
                set_living_standard = {
                    country = root
                    type = living_standard_utopian
                    cooldown = yes
                }
            }
        }
        # Noveria
        # Anhur
	}
}
