# Destroying a Planetary Station after Orbital Bombardment
planet_event = {
	id = planetarystations_action.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		planet_devastation >= 100
		has_planet_flag = planetary_station
	}

	immediate = {
		owner = { country_event = { id = planetarystations_action.2 days = 1 } }
		destroy_planetary_station = yes
	}
}

### Arrests and Transferring prisoners
planet_event = {
	id = planetarystations_action.3
	hide_window = yes
	
	is_triggered_only = yes
	
	trigger = {
		has_modifier = prison_station
		NOT = { has_modifier = planet_population_control }
		owner = {
			any_owned_planet = {
				OR = {
					any_owned_pop_group = { is_pop_category = criminal }
					AND = {
						has_modifier = political_persecution
						any_owned_pop_group = { has_ethic = ethic_egalitarian }
					}
					AND = {
						has_modifier = hunting_heretics
						any_owned_pop_group = { has_ethic = ethic_materialist }
					}
				}
			}
		}
	}
	
	immediate = {
		save_event_target_as = prison_station
		owner = {
			random_owned_planet = { 
				limit = { 
					OR = {
						any_owned_pop_group = { is_pop_category = criminal }
						AND = {
							has_modifier = political_persecution
							any_owned_pop_group = { has_ethic = ethic_egalitarian }
						}
						AND = {
							has_modifier = hunting_heretics
							any_owned_pop_group = { has_ethic = ethic_materialist }
						}
					}
				}
				if = {
					limit = { has_modifier = political_persecution }
					random_list = {
						50 = {
							weighted_random_owned_pop_group = {
								limit = { has_ethic = ethic_egalitarian }
								set_pop_convicted = yes
								
								export_trigger_value_to_variable = {
									trigger = pop_group_size
									variable = var_group_size
								}
								
								resettle_pop_group = {
									pop = this
									planet = event_target:prison_station
									PERCENTAGE = 1 # 100%
								}
								
								change_variable = {
									which = prev.var_convicted_pops
									value = var_group_size
								}
							}
							
							add_modifier = { 
								modifier = recent_arrests
								months = 3
							}
						}
						50 = {
							every_owned_pop_group = {
								limit = { has_ethic = ethic_egalitarian }
								set_pop_convicted = yes
								
								export_trigger_value_to_variable = {
									trigger = pop_group_size
									variable = var_group_size
								}
								
								resettle_pop_group = {
									pop = this
									planet = event_target:prison_station
									PERCENTAGE = 1 # 100%
								}
								
								change_variable = {
									which = prev.var_convicted_pops
									value = var_group_size
								}
							}
							
							add_modifier = { 
								modifier = mass_arrests
								months = 6
							}
						}
					}
				}
				else_if = {
					limit = { has_modifier = hunting_heretics }
					random_list = {
						50 = {
							weighted_random_owned_pop_group = {
								limit = { has_ethic = ethic_materialist }
								set_pop_convicted = yes
								
								export_trigger_value_to_variable = {
									trigger = pop_group_size
									variable = var_group_size
								}
								
								resettle_pop_group = {
									pop = this
									planet = event_target:prison_station
									PERCENTAGE = 1 # 100%
								}
								
								change_variable = {
									which = prev.var_convicted_pops
									value = var_group_size
								}
							}
							
							add_modifier = { 
								modifier = recent_arrests
								months = 3
							}
						}
						50 = {
							every_owned_pop_group = {
								limit = { has_ethic = ethic_materialist }
								set_pop_convicted = yes
								
								export_trigger_value_to_variable = {
									trigger = pop_group_size
									variable = var_group_size
								}
								
								resettle_pop_group = {
									pop = this
									planet = event_target:prison_station
									PERCENTAGE = 1 # 100%
								}
								
								change_variable = {
									which = prev.var_convicted_pops
									value = var_group_size
								}
							}
							
							add_modifier = { 
								modifier = mass_arrests
								months = 6
							}
						}
					}
				}
				else = {
					random_list = {
						50 = {
							weighted_random_owned_pop_group = {
								limit = {
									is_pop_category = criminal
								}
								set_pop_convicted = yes
								
								export_trigger_value_to_variable = {
									trigger = pop_group_size
									variable = var_group_size
								}
								
								resettle_pop_group = {
									pop = this
									planet = event_target:prison_station
									PERCENTAGE = 1 # 100%
								}
								
								change_variable = {
									which = prev.var_convicted_pops
									value = var_group_size
								}
							}
							
							add_modifier = { 
								modifier = recent_arrests
								months = 3
							}
						}
						50 = {
							every_owned_pop_group = {
								limit = {
									is_pop_category = criminal
								}
								set_pop_convicted = yes
								
								export_trigger_value_to_variable = {
									trigger = pop_group_size
									variable = var_group_size
								}
								
								resettle_pop_group = {
									pop = this
									planet = event_target:prison_station
									PERCENTAGE = 1 # 100%
								}
								
								change_variable = {
									which = prev.var_convicted_pops
									value = var_group_size
								}
							}
							
							add_modifier = { 
								modifier = mass_arrests
								months = 6
							}
						}
					}
				}
				
				create_message = {
					type = FELONY_TRANSPORT
					localization = FELONY_TRANSPORT_DESC
					days = 30
					target = THIS
					variable = {
						type = name
						localization = PLANET
						scope = this
					}
					variable = {
						type = name
						localization = PRISON
						scope = event_target:prison_station
					}
					variable = {
						type = variable
						varname = var_convicted_pops
						localization = VALUE
						scope = this
					}
				}
				
				set_variable = {
					which = var_convicted_pops
					value = 0
				}
			}
		}
	}
}

### Transferring served sentence pop
planet_event = {
	id = planetarystations_action.6
	hide_window = yes
	
	is_triggered_only = yes
	
	trigger = {
		has_modifier = prison_station
		any_owned_pop_group = {
			has_pop_group_flag = prisoners_flag
			NOT = { has_modifier = pop_convicted }
		}
	}
	
	immediate = {
		owner = {
			random_owned_planet = { 
				limit = { 
					NOR = {
						has_modifier = prison_station
						has_modifier = penal_colony
					}
				}
				save_event_target_as = transportation_planet
			}
		}
		every_owned_pop_group = {
			limit = { 
				has_pop_group_flag = prisoners_flag
				NOT = { has_modifier = pop_convicted }
			}
			
			remove_pop_group_flag = prisoners_flag
			
			export_trigger_value_to_variable = {
				trigger = pop_group_size
				variable = var_group_size
			}
		
			resettle_pop_group = {
				pop = this
				planet = event_target:transportation_planet
				PERCENTAGE = 1 # 100%
			}

			change_variable = {
				which = prev.var_prison_resettled_pops
				value = var_group_size
			}
		}
		
		create_message = {
			type = FREED_TRANSPORT
			localization = FREED_TRANSPORT_DESC
			days = 30
			target = THIS
			variable = {
				type = name
				localization = PLANET
				scope = event_target:transportation_planet
			}
			variable = {
				type = name
				localization = PRISON
				scope = root
			}
			variable = {
				type = variable
				varname = var_prison_resettled_pops
				localization = VALUE
				scope = this
			}
		}
		set_variable = {
			which = var_prison_resettled_pops
			value = 0
		}
	}
}

# Assigns station aging.
event = {
	id = planetarystations_action.8
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		every_galaxy_planet = {
			limit = {
				has_planet_flag = planetary_station
				is_colony = no
				NOT = { has_planet_flag = new_planet_station }
			}
			add_modifier = {
				modifier = "station_deterioration"
				years = 5
			}
			planet_event = { id = planetarystations_action.9 days = 1800 }
		}
	}
}

# The station became dilapidated and was demolished
planet_event = {
	id = planetarystations_action.9

		destroy_planetary_station = yes
	}

### Add Atmospheric Ecosystem
# on_survey
# Scope = Ship
# From = Planet
ship_event = {
	id = planetarystations_action.12
	title = "anomaly.300.name"
	desc = "anomaly.300.desc"
	picture = GFX_evt_atmospheric_flight
	show_sound = event_ship_bridge
	location = FROM

	is_triggered_only = yes

	trigger = {
		from = { has_planet_flag = gas_giant_life_deposit }
	}
	
	immediate = {
		from = { clear_deposits = yes }
		owner = { country_event = { id = story.2 days = 30 } }
	}

	option = {
		name = FASCINATING
		hidden_effect = {
			FROM = {
				add_modifier = {
					modifier = "atmospheric_ecosystem"
					days = -1
				}
			}
		}
		FROM = {
			set_deposit = d_society_3
		}
	}
}
