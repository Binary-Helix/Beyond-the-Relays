namespace = btr_terrorist


# Called during colony ship transit
fleet_event = {
    id = btr_terrorist.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        owner = { is_country_type = default }
        is_ship_class = shipclass_colonizer  # Colony ship
        NOT = { has_fleet_flag = terrorist_interdicted }
        is_in_frontier_space = yes # Ship in transit
    }

    immediate = {
        fleet.solar_system = {
            save_event_target_as = colonizer_last_known_location
        }
        random = { chance = 5 fleet_event = { id = btr_terrorist.10 } }
    }
}

# Stage 1: Colony Ship Goes Missing
fleet_event = {
    id = btr_terrorist.10
    title = btr_terrorist.10.name
    desc = btr_terrorist.10.desc
    picture = GFX_evt_ship_in_orbit
    show_sound = event_ship_thrusters

    is_triggered_only = yes

    immediate = {
        set_fleet_flag = terrorist_interdicted
    }

    option = {
        name = btr_terrorist.10.a  # "Write off the ship"

        destroy_fleet = this

        hidden_effect = {
            owner = {
                # Apply immediate losses
                add_resource = { influence = -50 }
                add_resource = { unity = -100 }

                # Political upheaval effects
                every_owned_planet = {
                    limit = { is_capital = yes }
                    add_modifier = {
                        modifier = colony_loss_unrest
                        years = 5
                    }
                }

              #  country_event = { id = btr_terrorist.99 }  # Cleanup event
            }
        }
    }

    option = {
        name = btr_terrorist.10.b  # "Authorize search and rescue"

        owner = {
            enable_special_project = {
                name = "PROJECT_MISSING_COLONY_SHIP_SEARCH"
                location = event_target:colonizer_last_known_location
                owner = this
            }
        }
    }
}

# Stage 2: Search Results - Hostage Situation
country_event = {
    id = btr_terrorist.20
    title = btr_terrorist.20.name
    desc = btr_terrorist.20.desc
    picture = GFX_evt_fleet_evil
    show_sound = event_ship_bridge

    is_triggered_only = yes

    option = {
        name = btr_terrorist.20.a  # "Pay the ransom - 1000 trade credits"

        trigger = {
            resource_stockpile_compare = { resource = trade value >= 1000 }
        }

        add_resource = { trade = -1000 }
        country_event = { id = btr_terrorist.30 }  # Ransom resolution
    }

    option = {
        name = btr_terrorist.20.b  # "Repeal anti-biotic policies"

        trigger = {
            has_policy_flag = research_restriction_dangerous_encouraged
        }

        set_policy = { policy = research_restriction_policy option = research_restriction_dangerous_forbidden }
        country_event = { id = btr_terrorist.40 }  # Policy change resolution
    }

    option = {
        name = btr_terrorist.20.c  # "Send in strike team"

        owner = {
            enable_special_project = {
                name = "PROJECT_MISSING_COLONY_SHIP_STRIKE_TEAM"
                location = event_target:colonizer_last_known_location
                owner = this
            }
        }
        #country_event = { id = btr_terrorist.50 }  # Boarding action
    }

    option = {
        name = btr_terrorist.20.d  # "We need more time"

        country_event = { id = btr_terrorist.21 days = 30 }  # Delay with risk
    }
}

# Delay Option - Risk of Terrorist Impatience
country_event = {
    id = btr_terrorist.21
    title = btr_terrorist.21.name
    desc = btr_terrorist.21.desc
    picture = GFX_evt_ship_in_orbit

    is_triggered_only = yes

    immediate = {
        random = {
            chance = 15  # 15% chance terrorists lose patience
            country_event = { id = btr_terrorist.60 }  # Ship destroyed
        }
    }

    # If ship not destroyed, return to main hostage event
    option = {
        name = btr_terrorist.21.a
        hidden_effect = {
            if = {
                limit = { NOT = { has_country_flag = ship_destroyed_by_terrorists } }
                country_event = { id = btr_terrorist.20 }
            }
        }
    }
}

# Stage 3A: Ransom Payment Resolution
country_event = {
    id = btr_terrorist.30
    title = btr_terrorist.30.name
    desc = {
        trigger = { NOT = { has_country_flag = ransom_betrayal } }
        text = btr_terrorist.30.desc.success
    }
    desc = {
        trigger = { has_country_flag = ransom_betrayal }
        text = btr_terrorist.30.desc.failure
    }
    picture = GFX_evt_ship_in_orbit

    is_triggered_only = yes

    immediate = {
        # Calculate success chance
        set_variable = { which = ransom_success_chance value = 60 }

        # Anti-biotic policy penalty
        if = {
            limit = { has_policy_flag = research_restriction_dangerous_encouraged }
            change_variable = { which = ransom_success_chance value = -10 }
        }

        # Roll for success
        random = {
            chance = ransom_success_chance
            # Success - ship returned
            set_country_flag = ransom_success
        }

        if = {
            limit = { NOT = { has_country_flag = ransom_success } }
            set_country_flag = ransom_betrayal
        }
    }

    option = {
        name = btr_terrorist.30.a

        if = {
            limit = { has_country_flag = ransom_success }
            # Positive outcome
            every_owned_planet = {
                limit = {
                    is_capital = yes
                }
                add_modifier = {
                    modifier = terrorist_minor_stability_reduction
                    years = 5
                }
            }
            change_variable = { which = btr_terrorist_modifier value = 1 }  # +1% future chance
        }
        else = {
            add_resource = { influence = -100 }
            change_variable = { which = btr_terrorist_modifier value = 2 }  # +2% future chance

            # Enhanced political upheaval
            every_owned_planet = {
                limit = {
                    is_capital = yes
                }
                add_modifier = {
                    modifier = terrorist_betrayal_unrest
                    years = 10
                }
                add_modifier = {
                    modifier = terrorist_stability_reduction
                    years = 5
                }
            }
        }

        hidden_effect = {
            #country_event = { id = btr_terrorist.99 }  # Cleanup
        }
    }
}

# Stage 3C: Boarding Action Resolution
country_event = {
    id = btr_terrorist.50
    title = btr_terrorist.50.name
    picture = GFX_btr_evt_biotic_terrorist

    is_triggered_only = yes

    desc = {
        trigger = { has_country_flag = boarding_complete_success }
        text = btr_terrorist.50.desc.complete_success
    }
    desc = {
        trigger = { has_country_flag = boarding_partial_success }
        text = btr_terrorist.50.desc.partial_success
    }
    desc = {
        trigger = { has_country_flag = boarding_pyrrhic_victory }
        text = btr_terrorist.50.desc.pyrrhic_victory
    }
    desc = {
        trigger = { has_country_flag = boarding_failure }
        text = btr_terrorist.50.desc.failure
    }

    immediate = {
        # Determine boarding outcome
        random_list = {
            30 = { set_country_flag = boarding_complete_success }
            40 = { set_country_flag = boarding_partial_success }
            20 = { set_country_flag = boarding_pyrrhic_victory }
            10 = { set_country_flag = boarding_failure }
        }
    }

    option = {
        name = btr_terrorist.50.a

        # Complete Success
        if = {
            limit = { has_country_flag = boarding_complete_success }
            add_resource = { influence = 50 }
            change_variable = { which = btr_terrorist_modifier value = -3 }  # -3% future chance
        }

        # Partial Success
        if = {
            limit = { has_country_flag = boarding_partial_success }
            change_variable = { which = btr_terrorist_modifier value = -1 }
            every_owned_planet = {
                limit = { is_capital = yes }
                add_modifier = {
                    modifier = terrorist_minor_stability_reduction
                    years = 1
                }
            }
            # Some colonist losses represented by resource penalty
            add_resource = { influence = -25 }
        }

        # Pyrrhic Victory
        if = {
            limit = { has_country_flag = boarding_pyrrhic_victory }
            add_resource = { influence = -50 }
            add_resource = { light_alloys = -25 }  # Ship repair costs
            add_resource = { heavy_alloys = -25 }  # Ship repair costs

            every_owned_planet = {
                limit = { is_capital = yes }
                add_modifier = {
                    modifier = military_casualties_mourning
                    months = 30
                }
                add_modifier = {
                    modifier = terrorist_reduced_stability_reduction
                    years = 1
                }
            }
        }

        # Complete Failure
        if = {
            limit = { has_country_flag = boarding_failure }
            add_resource = { influence = -100 }
            change_variable = { which = btr_terrorist_modifier value = 1 }

            every_owned_planet = {
                add_modifier = {
                    modifier = military_failure_unrest
                    years = 5
                }
                add_modifier = {
                    modifier = terrorist_stability_reduction
                    years = 5
                }
            }
        }

        hidden_effect = {
           # country_event = { id = btr_terrorist.99 }
        }
    }
}
